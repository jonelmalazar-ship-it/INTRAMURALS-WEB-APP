// ============================================================ // ACHS INTRAMURALS 2026 // SECURE BACKEND ENGINE // ============================================================ // // IMPORTANT: // Passwords are NOT stored in this source code. // // FIRST-TIME SETUP: // 1. Paste this entire script into Google Apps Script. // 2. Save. // 3. Run setupAdminPasswords() manually from the Apps Script editor. // 4. Enter the passwords when prompted. // 5. Authorize the script. // 6. Deploy / Update Web App. // // SECURITY: // - Passwords are stored as salted SHA-256 hashes. // - Admin authentication creates temporary server-side tokens. // - Edit / Publish / Audio operations require valid tokens. // - Passwords are never placed in URL parameters. // ============================================================ // ============================================================ // CONFIGURATION // ============================================================ var TOKEN_EXPIRY_MS = 8 * 60 * 60 * 1000; // 8 hours // ============================================================ // FIRST-TIME ADMIN PASSWORD SETUP // ============================================================ // // Run this manually from Apps Script editor. // The actual passwords are entered through prompts and are NOT // written into this source code. // // Four admin credentials are created: // // ADMIN // ADMIN1 // ADMIN2 // ADMIN3 // // ADMIN = normal operations / editing / publishing / audio // // ADMIN1 + ADMIN2 + ADMIN3 = final championship authorization // ============================================================ function setupAdminPasswords() { var ui = SpreadsheetApp.getUi(); var admin = ui.prompt( "ACHS SECURITY SETUP", "Enter NEW MAIN ADMIN password:", ui.ButtonSet.OK_CANCEL ); if (admin.getSelectedButton() != ui.Button.OK) { ui.alert("Setup cancelled."); return; } var admin1 = ui.prompt( "CHAMPIONSHIP ADMIN 1", "Enter password for Admin 1:", ui.ButtonSet.OK_CANCEL ); if (admin1.getSelectedButton() != ui.Button.OK) { ui.alert("Setup cancelled."); return; } var admin2 = ui.prompt( "CHAMPIONSHIP ADMIN 2", "Enter password for Admin 2:", ui.ButtonSet.OK_CANCEL ); if (admin2.getSelectedButton() != ui.Button.OK) { ui.alert("Setup cancelled."); return; } var admin3 = ui.prompt( "CHAMPIONSHIP ADMIN 3", "Enter password for Admin 3:", ui.ButtonSet.OK_CANCEL ); if (admin3.getSelectedButton() != ui.Button.OK) { ui.alert("Setup cancelled."); return; } var passwords = { ADMIN: admin.getResponseText(), ADMIN1: admin1.getResponseText(), ADMIN2: admin2.getResponseText(), ADMIN3: admin3.getResponseText() }; if ( !passwords.ADMIN || !passwords.ADMIN1 || !passwords.ADMIN2 || !passwords.ADMIN3 ) { ui.alert("ERROR: All four passwords are required."); return; } var props = PropertiesService.getScriptProperties(); Object.keys(passwords).forEach(function(role) { var salt = createRandomSalt(); var hash = hashPassword(passwords[role], salt); props.setProperty( "AUTH_" + role, JSON.stringify({ salt: salt, hash: hash }) ); }); ui.alert( "SECURITY SETUP COMPLETE", "All admin passwords have been securely hashed and stored in Script Properties.\n\n" + "The actual passwords are NOT stored in this script.", ui.ButtonSet.OK ); } // ============================================================ // PASSWORD FUNCTIONS // ============================================================ function createRandomSalt() { return Utilities.getUuid() + Utilities.getUuid(); } function hashPassword(password, salt) { var raw = String(salt) + "|" + String(password); var digest = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, raw, Utilities.Charset.UTF_8 ); return digest.map(function(byte) { var value = byte; if (value < 0) { value += 256; } var hex = value.toString(16); return hex.length === 1 ? "0" + hex : hex; }).join(""); } function verifyPassword(role, password) { var props = PropertiesService.getScriptProperties(); var stored = props.getProperty("AUTH_" + role); if (!stored) { return false; } try { var data = JSON.parse(stored); var incomingHash = hashPassword(password, data.salt); return incomingHash === data.hash; } catch (err) { return false; } } // ============================================================ // TOKEN FUNCTIONS // ============================================================ function createAuthToken(role) { var props = PropertiesService.getScriptProperties(); var token = Utilities.getUuid().replace(/-/g, "") + Utilities.getUuid().replace(/-/g, ""); var tokenData = { role: role, created: Date.now(), expires: Date.now() + TOKEN_EXPIRY_MS }; props.setProperty( "TOKEN_" + token, JSON.stringify(tokenData) ); return token; } function getTokenData(token) { if (!token) { return null; } var props = PropertiesService.getScriptProperties(); var raw = props.getProperty("TOKEN_" + token); if (!raw) { return null; } try { var data = JSON.parse(raw); if (!data.expires || Date.now() > data.expires) { props.deleteProperty("TOKEN_" + token); return null; } return data; } catch (err) { return null; } } function requireToken(token, allowedRoles) { var data = getTokenData(token); if (!data) { throw new Error("AUTH_REQUIRED"); } if ( allowedRoles && allowedRoles.indexOf(data.role) === -1 ) { throw new Error("ACCESS_DENIED"); } return data; } function cleanupExpiredTokens() { var props = PropertiesService.getScriptProperties(); var all = props.getProperties(); Object.keys(all).forEach(function(key) { if (key.indexOf("TOKEN_") !== 0) { return; } try { var data = JSON.parse(all[key]); if (!data.expires || Date.now() > data.expires) { props.deleteProperty(key); } } catch (err) { props.deleteProperty(key); } }); } // ============================================================ // JSON RESPONSE HELPER // ============================================================ function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // GET REQUEST // ============================================================ // // PUBLIC DATA ONLY. // // IMPORTANT: // No passwords. // No admin credentials. // No tokens. // No private information. // // ============================================================ function doGet(e) { var params = (e && e.parameter) || {}; // If this GET request carries an admin action (or audioStatus), // route it through the same handler doPost uses. GET responses // are readable by the browser (unlike POST responses sent with // mode:"no-cors"), so this lets the frontend see the real // success/error/token instead of guessing. if (params.action || params.audioStatus) { return handleAction(params); } var ss = SpreadsheetApp.getActiveSpreadsheet(); // ---------------------------------------------------------- // STANDINGS // ---------------------------------------------------------- var standingsSheet = ss.getSheetByName("Standings"); var standingsData = []; if (standingsSheet) { var sValues = standingsSheet.getDataRange().getValues(); for (var i = 1; i < sValues.length; i++) { if (sValues[i][0] !== "") { standingsData.push({ faction: sValues[i][0], gold: Number(sValues[i][1]) || 0, silver: Number(sValues[i][2]) || 0, bronze: Number(sValues[i][3]) || 0 }); } } } // ---------------------------------------------------------- // EVENTS // ---------------------------------------------------------- var eventsSheet = ss.getSheetByName("Events"); var eventsData = []; if (eventsSheet) { var eValues = eventsSheet.getDataRange().getValues(); for (var j = 1; j < eValues.length; j++) { if ( eValues[j][0] !== "" || eValues[j][1] !== "" ) { eventsData.push({ category: eValues[j][0], event: eValues[j][1], gold: eValues[j][2], silver: eValues[j][3], bronze: eValues[j][4] }); } } } // ---------------------------------------------------------- // AUDIO // ---------------------------------------------------------- var props = PropertiesService.getScriptProperties(); var currentAudioStatus = props.getProperty("audioStatus") || "pause"; var currentAudioTrack = props.getProperty("audioTrack") || ""; // ---------------------------------------------------------- // PUBLIC RESPONSE // ---------------------------------------------------------- return jsonResponse({ standings: standingsData, events: eventsData, audio: { status: currentAudioStatus, track: currentAudioTrack } }); } // ============================================================ // POST REQUEST // ============================================================ // // Kept for backward compatibility. Delegates to the same handler // doGet's action branch uses. // ============================================================ function doPost(e) { var params = (e && e.parameter) || {}; return handleAction(params); } // ============================================================ // SHARED ADMIN ACTION HANDLER // ============================================================ // // This is the original doPost() body, extracted into its own // function so both doGet (for readable responses) and doPost // can call it. Nothing about the login/audio/publish/edit/ // champion logic changed - "delete" is newly added at the end. // ============================================================ function handleAction(params) { try { var action = params.action || ""; // ======================================================== // LOGIN // ======================================================== if (action === "login") { cleanupExpiredTokens(); var role = String(params.role || "").toUpperCase(); var password = String(params.password || ""); var allowedRoles = [ "ADMIN", "ADMIN1", "ADMIN2", "ADMIN3" ]; if (allowedRoles.indexOf(role) === -1) { return jsonResponse({ status: "error", message: "Invalid role." }); } if (!password) { return jsonResponse({ status: "error", message: "Password required." }); } if (!verifyPassword(role, password)) { return jsonResponse({ status: "error", message: "Authentication failed." }); } var token = createAuthToken(role); return jsonResponse({ status: "success", role: role, token: token, expiresIn: TOKEN_EXPIRY_MS }); } // ======================================================== // LOGOUT // ======================================================== if (action === "logout") { var logoutToken = params.token || ""; if (logoutToken) { PropertiesService .getScriptProperties() .deleteProperty("TOKEN_" + logoutToken); } return jsonResponse({ status: "success" }); } // ======================================================== // AUDIO BROADCAST // ======================================================== // // MAIN ADMIN ONLY // // ======================================================== if (params.audioStatus) { requireToken( params.token, ["ADMIN"] ); var propsAudio = PropertiesService.getScriptProperties(); var audioStatus = params.audioStatus === "play" ? "play" : "pause"; propsAudio.setProperty( "audioStatus", audioStatus ); if (params.audioTrack !== undefined) { propsAudio.setProperty( "audioTrack", params.audioTrack || "" ); } return jsonResponse({ status: "audio_success" }); } // ======================================================== // PUBLISH MATCH RESULT // ======================================================== // // MAIN ADMIN ONLY // // ======================================================== if ( action === "publish" || (!action && params.event) ) { requireToken( params.token, ["ADMIN"] ); publishEventResult(params); return jsonResponse({ status: "success", message: "Result saved." }); } // ======================================================== // EDIT EVENT RESULT // ======================================================== // // MAIN ADMIN ONLY // // ======================================================== if (action === "edit") { requireToken( params.token, ["ADMIN"] ); var editIndex = parseInt(params.index, 10); if (isNaN(editIndex)) { return jsonResponse({ status: "error", message: "Invalid event index." }); } var editResult = editEventResult( editIndex, params.event || "", params.gold || "", params.silver || "", params.bronze || "" ); return jsonResponse(editResult); } // ======================================================== // DELETE EVENT RESULT // ======================================================== // // MAIN ADMIN ONLY // // ======================================================== if (action === "delete") { requireToken( params.token, ["ADMIN"] ); var deleteIndex = parseInt(params.index, 10); if (isNaN(deleteIndex)) { return jsonResponse({ status: "error", message: "Invalid event index." }); } var deleteResult = deleteEventResult(deleteIndex); return jsonResponse(deleteResult); } // ======================================================== // CHAMPIONSHIP AUTHORIZATION // ======================================================== // // IMPORTANT: // // Each admin must authenticate separately. // // The frontend sends the three temporary admin tokens. // // The backend verifies that: // // token 1 = ADMIN1 // token 2 = ADMIN2 // token 3 = ADMIN3 // // ======================================================== if (action === "champion") { var token1 = requireToken( params.token1, ["ADMIN1"] ); var token2 = requireToken( params.token2, ["ADMIN2"] ); var token3 = requireToken( params.token3, ["ADMIN3"] ); return jsonResponse({ status: "success", authorized: true, message: "Three-admin championship authorization successful." }); } return jsonResponse({ status: "error", message: "Unknown or unauthorized action." }); } catch (err) { var message = String(err.message || err); if (message === "AUTH_REQUIRED") { return jsonResponse({ status: "error", code: "AUTH_REQUIRED", message: "Valid administrator authentication is required." }); } if (message === "ACCESS_DENIED") { return jsonResponse({ status: "error", code: "ACCESS_DENIED", message: "You are not authorized to perform this operation." }); } return jsonResponse({ status: "error", message: message }); } } // ============================================================ // PUBLISH EVENT RESULT // ============================================================ function publishEventResult(params) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var category = params.category || "General Sports"; var eventName = params.event || ""; var gold = params.gold || ""; var silver = params.silver || ""; var bronze = params.bronze || ""; if (!eventName) { throw new Error( "Missing event name." ); } // ---------------------------------------------------------- // EVENTS SHEET // ---------------------------------------------------------- var eventsSheet = ss.getSheetByName("Events"); if (!eventsSheet) { eventsSheet = ss.insertSheet("Events"); eventsSheet.appendRow([ "Category", "Event", "Gold", "Silver", "Bronze" ]); } eventsSheet.appendRow([ category, eventName, gold, silver, bronze ]); // ---------------------------------------------------------- // STANDINGS SHEET // ---------------------------------------------------------- var standingsSheet = ss.getSheetByName("Standings"); if (!standingsSheet) { standingsSheet = ss.insertSheet("Standings"); standingsSheet.appendRow([ "Faction", "Gold", "Silver", "Bronze" ]); var defaultFactions = [ "Moniyan Empire", "Eruditio", "Abyssal Domain", "Cadia Riverlands" ]; for ( var f = 0; f < defaultFactions.length; f++ ) { standingsSheet.appendRow([ defaultFactions[f], 0, 0, 0 ]); } } updateStandings( standingsSheet, gold, silver, bronze ); } // ============================================================ // UPDATE STANDINGS // ============================================================ function updateStandings( standingsSheet, gold, silver, bronze ) { var standingsData = standingsSheet .getDataRange() .getValues(); for ( var r = 1; r < standingsData.length; r++ ) { var faction = standingsData[r][0]; var currentGold = Number(standingsData[r][1]) || 0; var currentSilver = Number(standingsData[r][2]) || 0; var currentBronze = Number(standingsData[r][3]) || 0; if (faction === gold) { currentGold++; } if (faction === silver) { currentSilver++; } if (faction === bronze) { currentBronze++; } standingsSheet .getRange(r + 1, 2) .setValue(currentGold); standingsSheet .getRange(r + 1, 3) .setValue(currentSilver); standingsSheet .getRange(r + 1, 4) .setValue(currentBronze); } } // ============================================================ // EDIT EVENT RESULT // ============================================================ // // IMPORTANT: // // We don't simply change the event row. // We recalculate the entire standings sheet from Events. // // This prevents medal counts from becoming incorrect after // an edit. // // ============================================================ function editEventResult( index, eventName, gold, silver, bronze ) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var eventsSheet = ss.getSheetByName("Events"); if (!eventsSheet) { throw new Error( "Events sheet not found." ); } var row = index + 2; if ( row < 2 || row > eventsSheet.getLastRow() ) { throw new Error( "Invalid event record." ); } eventsSheet .getRange(row, 2) .setValue(eventName); eventsSheet .getRange(row, 3) .setValue(gold); eventsSheet .getRange(row, 4) .setValue(silver); eventsSheet .getRange(row, 5) .setValue(bronze); rebuildStandings(); return { status: "success", message: "Event result updated successfully." }; } // ============================================================ // DELETE EVENT RESULT // ============================================================ // // Removes the row entirely from the Events sheet, then, same // as editEventResult(), recalculates the whole Standings sheet // from what's left in Events - so medal counts stay correct // after a deletion too. // ============================================================ function deleteEventResult(index) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var eventsSheet = ss.getSheetByName("Events"); if (!eventsSheet) { throw new Error( "Events sheet not found." ); } var row = index + 2; if ( row < 2 || row > eventsSheet.getLastRow() ) { throw new Error( "Invalid event record." ); } eventsSheet.deleteRow(row); rebuildStandings(); return { status: "success", message: "Event result deleted successfully." }; } // ============================================================ // REBUILD STANDINGS FROM EVENTS // ============================================================ function rebuildStandings() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var eventsSheet = ss.getSheetByName("Events"); var standingsSheet = ss.getSheetByName("Standings"); if (!eventsSheet) { return; } if (!standingsSheet) { standingsSheet = ss.insertSheet("Standings"); } var factions = [ "Moniyan Empire", "Eruditio", "Abyssal Domain", "Cadia Riverlands" ]; var tally = {}; factions.forEach(function(faction) { tally[faction] = { gold: 0, silver: 0, bronze: 0 }; }); var values = eventsSheet .getDataRange() .getValues(); for ( var i = 1; i < values.length; i++ ) { var gold = String(values[i][2] || "").trim(); var silver = String(values[i][3] || "").trim(); var bronze = String(values[i][4] || "").trim(); if (tally[gold]) { tally[gold].gold++; } if (tally[silver]) { tally[silver].silver++; } if (tally[bronze]) { tally[bronze].bronze++; } } standingsSheet.clearContents(); standingsSheet.appendRow([ "Faction", "Gold", "Silver", "Bronze" ]); factions.forEach(function(faction) { standingsSheet.appendRow([ faction, tally[faction].gold, tally[faction].silver, tally[faction].bronze ]); }); } // ============================================================ // OPTIONAL ADMIN STATUS CHECK // ============================================================ function validateToken(token) { var data = getTokenData(token); if (!data) { return { valid: false }; } return { valid: true, role: data.role, expires: data.expires }; }
